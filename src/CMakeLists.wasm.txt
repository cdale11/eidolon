# WASM build configuration for Eidolon ReplicaCore
# This builds a portable simulation core that can run in browsers via WebAssembly

cmake_minimum_required(VERSION 3.20)
project(eidolon_wasm LANGUAGES CXX)

# Use the emscripten toolchain
if(NOT CMAKE_TOOLCHAIN_FILE)
  message(FATAL_ERROR "Must use -DCMAKE_TOOLCHAIN_FILE=../cmake/emscripten.cmake")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Build configuration
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O3 -DNDEBUG")
# Disable FP contraction (FMA) so WASM and native execute identical IEEE
# arithmetic — required for bit-exact native/WASM parity (see core/detmath).
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -ffp-contract=off")

# WASM-specific definitions
add_compile_definitions(
  EIDOLON_WASM_BUILD=1
  EIDOLON_NO_SQLITE=1
  EIDOLON_NO_THREADS=1
)

# Portable ReplicaCore: core simulation without platform dependencies
# This includes: RNG, clock, world, physiology, mind (learning, memory, planning, etc.)
# Excludes: SQLite archive, LLM bridge, server code

add_library(eidolon_replica_core STATIC
  core/rng.cpp
  core/log.cpp
  core/serialize.cpp
  core/json.cpp
  core/detmath.cpp
  world/world.cpp
  world/wildlife.cpp
  world/cellular_automata.cpp
  world/voronoi.cpp
  world/reaction_diffusion.cpp
  world/boids.cpp
  world/markov.cpp
  world/lsystem.cpp
  world/procgen.cpp
  world/grammar.cpp
  body/physiology.cpp
  body/ode_tests.cpp
  body/skill.cpp
  body/crafting.cpp
  body/affordance.cpp
  body/construction.cpp
  mind/memory.cpp
  mind/memory_system.cpp
  mind/mlp.cpp
  mind/neuromod.cpp
  mind/value_net.cpp
  mind/threat_net.cpp
  mind/policy.cpp
  mind/attention.cpp
  mind/personality.cpp
  mind/learn.cpp
  mind/world_predictor.cpp
  mind/goal_emergence.cpp
  mind/llm_planner.cpp
  mind/user_model.cpp
  mind/wildlife_social.cpp
  mind/attachment.cpp
  mind/belief_ising.cpp
  mind/markov.cpp
  mind/self_model.cpp
  mind/metacognition.cpp
  mind/concept_formation.cpp
  mind/graph_rewriting.cpp
  mind/reflection.cpp
  mind/grounded_language.cpp
  mind/compute_profile.cpp
  mind/compute_scheduler.cpp
  mind/heredity.cpp
  sim/engine.cpp
)

target_include_directories(eidolon_replica_core PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
)

# WASM-specific compile options (no linker flags here)
target_compile_options(eidolon_replica_core PRIVATE
  -fno-exceptions
  -fno-rtti
  -std=c++17
)

# Test executable to verify WASM linking works
add_executable(eidolon_wasm_test
  ../tools/eidolon-wasm-test.cpp
)
target_link_libraries(eidolon_wasm_test PRIVATE eidolon_replica_core)

# Phase 12 parity harness (WASM side): same source as the native eidolon-parity-dump.
# Compiled under node so `node eidolon_parity_dump.js --seed 42 --ticks 1000` prints the
# same `parity:` digest line as the native binary when the core is bit-for-bit portable.
add_executable(eidolon_parity_dump
  ../tools/eidolon-parity-dump.cpp
)
target_link_libraries(eidolon_parity_dump PRIVATE eidolon_replica_core)
# Plain node executable (no MODULARIZE factory): main() runs under `node <file>.js`
# and prints the `parity:` digest line that the native binary also prints.
target_link_options(eidolon_parity_dump PRIVATE
  -sWASM=1
  -sMODULARIZE=0
  -sALLOW_MEMORY_GROWTH=1
  -sENVIRONMENT=node
  -sNODERAWFS=1
  -sINITIAL_MEMORY=64MB
  -sMAXIMUM_MEMORY=256MB
  -sSTACK_SIZE=5MB
)

# WASM link options for executable
target_link_options(eidolon_wasm_test PRIVATE
  -sWASM=1
  -sALLOW_MEMORY_GROWTH=1
  -sEXPORT_ES6=0
  -sMODULARIZE=1
  -sEXPORT_NAME=EidolonReplicaCore
  -sENVIRONMENT=web,worker,node
  -sINITIAL_MEMORY=64MB
  -sMAXIMUM_MEMORY=256MB
  -sSTACK_SIZE=5MB
  -sASSERTIONS=0
  -sSAFE_HEAP=0
)

# Export compile commands for tooling
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
